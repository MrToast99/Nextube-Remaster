cmake_minimum_required(VERSION 3.16)

set(EXTRA_COMPONENT_DIRS "components")

# ── Read firmware version from version.json ──────────────────────────
# Tell CMake to re-run configure automatically whenever version.json changes.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/version.json")

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/version.json" _VER_JSON)
# Use a robust semver pattern rather than [[:space:]] (POSIX classes can
# misbehave on Windows where file(READ) preserves CRLF line endings).
string(REGEX MATCH "\"firmware_version\"[^\"]*\"([0-9]+[.][0-9]+[.][0-9]+)\"" _ "${_VER_JSON}")
set(FW_VERSION "${CMAKE_MATCH_1}")
if(NOT FW_VERSION)
    message(WARNING "Could not parse firmware_version from version.json – defaulting to 0.0.0")
    set(FW_VERSION "0.0.0")
endif()

string(REGEX MATCH "\"fs_version\"[^\"]*\"([0-9]+[.][0-9]+[.][0-9]+)\"" _ "${_VER_JSON}")
set(FS_VERSION "${CMAKE_MATCH_1}")
if(NOT FS_VERSION)
    message(WARNING "Could not parse fs_version from version.json – defaulting to ${FW_VERSION}")
    set(FS_VERSION "${FW_VERSION}")
endif()

# Write LittleFS version into the web tree for runtime mismatch detection.
# The firmware reports FW_VERSION_STR; the web UI reports this file's content.
# Bumping only fs_version (data/ changes) never triggers the mismatch banner.
# Bumping only firmware_version (code changes) will trigger it until littlefs.bin is re-flashed.
file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/data/web/version.txt" "${FS_VERSION}")

# Generate fw_version.h as a #ifndef-guarded header (belt-and-suspenders
# fallback in case a translation unit includes it without the -D flag).
# The authoritative injection is target_compile_definitions(PUBLIC) in
# components/config_mgr/CMakeLists.txt — that propagates to every consumer.
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/main/fw_version.h.in"
    "${CMAKE_CURRENT_SOURCE_DIR}/components/config_mgr/include/fw_version.h"
    @ONLY
)

# ── Generate the "stock files" manifest ────────────────────────────────
# Every file under data/ as of this build (config.json excluded — that's
# user data, never stock) becomes an expected-present entry that
# web_server.c's boot-time stock-file integrity check verifies against
# LittleFS at runtime — see the STOCK_FILES doc comment there. CONFIGURE_DEPENDS
# re-globs on every build so a file added/removed under data/ is picked up
# without needing to touch this CMakeLists.txt, the same way version.json's
# dependency above works.
#
# web/index.html is deliberately excluded: the CI hotpatch tooling
# (.github/scripts/make_Webui_patch.py's GZIP_ONLY set) converts it to
# index.html.gz on every device that's ever applied a hotpatch, and
# hp_drop_stale_index() (web_server.c) deliberately DELETES the plain copy
# once the .gz exists — so "index.html present" is actually a sign of a
# device that has NEVER been updated, not a healthy-device signal, and
# neither path alone is a reliable stock-file check. The runtime check
# handles this pair as a special either/or case instead (see web_server.c).
file(GLOB_RECURSE _STOCK_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/data"
     CONFIGURE_DEPENDS
     "${CMAKE_CURRENT_SOURCE_DIR}/data/*")
list(REMOVE_ITEM _STOCK_FILES "config.json")
list(REMOVE_ITEM _STOCK_FILES "web/index.html")
list(REMOVE_ITEM _STOCK_FILES "images/album/.keep")   # git placeholder, not a real stock asset
list(SORT _STOCK_FILES)

# Each file's SHA-256 rides alongside its path (STOCK_SHA256[i] matches
# STOCK_FILES[i]) so the device's OWN firmware image already carries what
# every stock file's content is supposed to be — no manifest ever needs to
# be fetched over the network. Used by web_server.c's live-source repair
# (stock_repair_task()): a device that finds a file missing or its local
# hash not matching the compiled-in one fetches ONLY that specific file,
# from the git tag matching this firmware's OWN fs_version, and re-verifies
# the download against this same hash before writing it. This is why
# per-file content hashing, not just presence, matters here — presence
# alone (STOCK_FILES on its own) can't tell a stale file from a current one.
# file(SHA256) is a built-in CMake command (no external script needed).

set(_STOCK_FILES_H "/* Auto-generated by CMake (this file) from the data/ tree at build time --\n")
string(APPEND _STOCK_FILES_H " * do not edit by hand, it will be overwritten on the next configure. See\n")
string(APPEND _STOCK_FILES_H " * the STOCK_FILES doc comment in web_server.c for what this is used for,\n")
string(APPEND _STOCK_FILES_H " * and the CMakeLists.txt comment above the generation code for why\n")
string(APPEND _STOCK_FILES_H " * web/index.html is deliberately excluded here. STOCK_SHA256[i] is the\n")
string(APPEND _STOCK_FILES_H " * SHA-256 of STOCK_FILES[i]'s content at build time, hex-encoded — see\n")
string(APPEND _STOCK_FILES_H " * the comment above the generation code for what it's for. */\n")
string(APPEND _STOCK_FILES_H "#pragma once\n\n")
string(APPEND _STOCK_FILES_H "static const char *const STOCK_FILES[] = {\n")
foreach(_f ${_STOCK_FILES})
    string(APPEND _STOCK_FILES_H "    \"${_f}\",\n")
endforeach()
string(APPEND _STOCK_FILES_H "};\n")
string(APPEND _STOCK_FILES_H "static const char *const STOCK_SHA256[] = {\n")
foreach(_f ${_STOCK_FILES})
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/data/${_f}" _hash)
    string(APPEND _STOCK_FILES_H "    \"${_hash}\",\n")
endforeach()
string(APPEND _STOCK_FILES_H "};\n")
string(APPEND _STOCK_FILES_H "#define STOCK_FILES_COUNT (sizeof(STOCK_FILES) / sizeof(STOCK_FILES[0]))\n")

file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/components/config_mgr/include/stock_files.h" "${_STOCK_FILES_H}")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

project(nextube-fw)

message(STATUS "Nextube-Remaster firmware v${FW_VERSION}  LittleFS v${FS_VERSION}")

# Embed the LittleFS web filesystem into the firmware
littlefs_create_partition_image(littlefs data FLASH_IN_PROJECT)
